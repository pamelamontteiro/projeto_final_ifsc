from pyodide.http import pyfetch import asyncio import json endpoint = "http://localhost:8000/usuarios/" async def cadastro_usuario(*args, **kwargs): nome = Element('nome').value email = Element('email').value senha = Element('senha').value data = {"nome": nome, "email": email, "tipo": 2, "senha": senha} data_str = json.dumps(data, ensure_ascii=True) response = await pyfetch( endpoint, method="POST", headers={"Content-Type": "application/json"}, body=f"{data_str}", mode="no-cors" )
